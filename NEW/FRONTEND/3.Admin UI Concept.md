```tsx
  LayoutDashboard,

const links = [
  { name: "Dashboard", to: "/admin", icon: LayoutDashboard },
  { name: "Nodes", to: "/admin/nodes", icon: Server },
  { name: "Users", to: "/admin/users", icon: Users },
  { name: "Jobs", to: "/admin/jobs", icon: Briefcase },
  { name: "Resource Requests", to: "/admin/requests", icon: ClipboardList },
  { name: "Audit Logs", to: "/admin/audit", icon: FileText },

export function AdminSidebar() {
  return (
    <aside className="hidden md:flex w-64 flex-col border-r bg-white">
      <div className="h-16 flex items-center px-6 font-semibold text-lg">
        Compute Admin
      </div>

      <nav className="flex-1 space-y-1 px-3">
        {links.map(({ name, to, icon: Icon }) => (
          <NavLink
            key={name}
            to={to}
            className={({ isActive }) =>
              `flex items-center gap-3 rounded-md px-3 py-2 text-sm font-medium transition
               ${
                 isActive
                   ? "bg-slate-100 text-slate-900"
                   : "text-slate-600 hover:bg-slate-50"
               }`
            }
          >
            <Icon className="h-4 w-4" />
            {name}
          </NavLink>
        ))}
      </nav>
    </aside>
  )
}
```

## 🎛️ 2. Admin Topbar

### `src/layouts/admin/AdminTopbar.tsx`

```tsx
import { LogOut } from "lucide-react"
import { Button } from "@/components/ui/button"
import { useAuthStore } from "@/store/auth.store"
import { useNavigate } from "react-router-dom"

export function AdminTopbar({ title }: { title: string }) {
  const logout = useAuthStore((s) => s.logout)
  const navigate = useNavigate()

  return (
    <header className="h-16 border-b bg-white flex items-center justify-between px-6">
      <h1 className="text-lg font-semibold text-slate-800">
        {title}
      </h1>

      <Button
        variant="ghost"
        onClick={() => {
          logout()
          navigate("/login")
        }}
        className="flex items-center gap-2"
      >
        <LogOut className="h-4 w-4" />
        Logout
      </Button>
    </header>
  )
}
```

## 🧩 3. Admin Layout (Sidebar + Topbar + Outlet)

### `src/layouts/admin/AdminLayout.tsx`

```tsx
import { Outlet, useLocation } from "react-router-dom"
import { AdminSidebar } from "./AdminSidebar"
import { AdminTopbar } from "./AdminTopbar"

const titleMap: Record<string, string> = {
  "/admin": "Dashboard",
  "/admin/nodes": "Nodes",
  "/admin/users": "Users",
  "/admin/jobs": "Jobs",
  "/admin/requests": "Resource Requests",
  "/admin/audit": "Audit Logs",
}

export default function AdminLayout() {
  const location = useLocation()
  const title = titleMap[location.pathname] || "Admin"

  return (
    <div className="min-h-screen flex bg-slate-50">
      <AdminSidebar />

      <div className="flex flex-col flex-1">
        <AdminTopbar title={title} />

        <main className="p-6">
          <Outlet />
        </main>
      </div>
    </div>
  )
}
```
