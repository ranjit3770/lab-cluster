# 3️⃣ USERS PAGE (GLOBAL / NODE ASSIGNMENT UI)

This is the **most critical admin screen**.

### UI Rules (Strict)

* Admin chooses **assignment mode**
* GLOBAL → CPU + RAM only
* NODE → select nodes + per-node quota
* Validation happens backend-side

---

### 📁 `src/pages/admin/UsersPage.tsx`

```tsx
import { useState } from "react"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Select, SelectItem, SelectTrigger, SelectContent, SelectValue } from "@/components/ui/select"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"

export default function UsersPage() {
  const [mode, setMode] = useState<"GLOBAL" | "NODE">("GLOBAL")

  return (
    <Card className="max-w-2xl">
      <CardHeader>
        <CardTitle>Create / Edit User</CardTitle>
      </CardHeader>

      <CardContent className="space-y-4">
        <Input placeholder="Username" />

        <Select value={mode} onValueChange={(v) => setMode(v as any)}>
          <SelectTrigger>
            <SelectValue placeholder="Assignment Mode" />
          </SelectTrigger>
          <SelectContent>
            <SelectItem value="GLOBAL">GLOBAL</SelectItem>
            <SelectItem value="NODE">NODE</SelectItem>
          </SelectContent>
        </Select>

        {mode === "GLOBAL" && (
          <div className="grid grid-cols-2 gap-4">
            <Input placeholder="CPU" type="number" />
            <Input placeholder="Memory (GB)" type="number" />
          </div>
        )}

        {mode === "NODE" && (
          <div className="space-y-2">
            <p className="text-sm text-slate-500">
              Assign per-node quota
            </p>
            <div className="grid grid-cols-3 gap-2">
              <Input value="node-1" disabled />
              <Input placeholder="CPU" type="number" />
              <Input placeholder="Memory (GB)" type="number" />
            </div>
          </div>
        )}

        <Button className="w-full">Save User</Button>
      </CardContent>
    </Card>
  )
}
```

✔ Enforces **FR-3, FR-4, FR-5**
✔ No user-side infra exposure

### Admin Job Monitoring

* Status badge

---

### 📁 `src/pages/admin/JobsPage.tsx`

```tsx
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow
} from "@/components/ui/table"
import { Badge } from "@/components/ui/badge"

const jobs = [
  {
    id: 101,
    user: "alice",
    status: "RETRY_WAITING",
    retries: 1,
    failure: "INFRA",
  },
]

export default function JobsPage() {
  return (
    <div className="space-y-4">
      <h2 className="text-xl font-semibold">All Jobs</h2>

      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>ID</TableHead>
            <TableHead>User</TableHead>
            <TableHead>Status</TableHead>
            <TableHead>Retries</TableHead>
            <TableHead>Failure</TableHead>
          </TableRow>
        </TableHeader>

        <TableBody>
          {jobs.map((j) => (
            <TableRow key={j.id}>
              <TableCell>{j.id}</TableCell>
              <TableCell>{j.user}</TableCell>
              <TableCell>
                <Badge variant={
                  j.status === "COMPLETED"
                    ? "success"
                    : j.status === "FAILED"
                    ? "destructive"
                    : "warning"
                }>
                  {j.status}
                </Badge>
              </TableCell>
              <TableCell>{j.retries}</TableCell>
              <TableCell>{j.failure || "-"}</TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>
    </div>
  )
}
```

✔ Reflects **FR-10 → FR-12**
✔ Infra vs User failure visible only to admin
