# 1️⃣ API INTEGRATION — Jobs & Dashboard

## 📁 `src/services/jobs.service.ts`

```ts
import { api } from "./api"

export interface Job {
  id: number
  status: string
  retry_count: number
}

export async function submitJob(
  command: string,
  argumentsList: string[]
) {
  const { data } = await api.post("/jobs", {
    command,
    arguments: argumentsList,
  })
  return data
}

export async function fetchMyJobs(): Promise<Job[]> {
  const { data } = await api.get("/jobs")
  return data
}

export async function cancelJob(jobId: number) {
  await api.delete(`/jobs/${jobId}`)
}
```

# 2️⃣ DASHBOARD — Real API + Loading

## 📁 `UserDashboard.tsx` (updated)

import { useEffect, useState } from "react"

    fetchResourceHealth()
      .then((res) => setState(res.resource_state))
  }, [])

        <CardTitle>Resource Status</CardTitle>

      <CardContent>

              : state === "IN_USE"
              ? "warning"
              : "destructive"
          }
        >
          {state}
        </Badge>
      </CardContent>

# 3️⃣ JOB SUBMIT — API + TOAST

## 📁 `JobSubmitCard.tsx` (updated)

```tsx
import { useState } from "react"
import { submitJob } from "@/services/jobs.service"
import { useToast } from "@/components/ui/use-toast"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Textarea } from "@/components/ui/textarea"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"

export function JobSubmitCard({ onSubmitted }: { onSubmitted: () => void }) {
  const [command, setCommand] = useState("")
  const [args, setArgs] = useState("")
  const [loading, setLoading] = useState(false)
  const { toast } = useToast()

  const handleSubmit = async () => {
    if (!command) {
      toast({ variant: "destructive", title: "Command required" })
      return
    }

    setLoading(true)
    try {
      await submitJob(command, args.split(" ").filter(Boolean))
      toast({ title: "Job submitted successfully" })
      setCommand("")
      setArgs("")
      onSubmitted()
    } catch {
      toast({
        variant: "destructive",
        title: "Failed to submit job",
      })
    } finally {
      setLoading(false)
    }
  }

  return (
    <Card>
      <CardHeader>
        <CardTitle>Submit Job</CardTitle>
      </CardHeader>
      <CardContent className="space-y-3">
        <Input
          value={command}
          onChange={(e) => setCommand(e.target.value)}
          placeholder="python train.py"
        />
        <Textarea
          value={args}
          onChange={(e) => setArgs(e.target.value)}
          placeholder="--epochs 10"
        />
        <Button
          className="w-full"
          onClick={handleSubmit}
          disabled={loading}
        >
          {loading ? "Submitting..." : "Submit"}
        </Button>
      </CardContent>
    </Card>
  )
}
```

# 4️⃣ JOB LIST — API + CANCEL + EMPTY STATE

## 📁 `JobList.tsx` (updated)

```tsx
import { useEffect, useState } from "react"
import { fetchMyJobs, cancelJob, Job } from "@/services/jobs.service"
import { Badge } from "@/components/ui/badge"
import { Button } from "@/components/ui/button"
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow
} from "@/components/ui/table"
import { Skeleton } from "@/components/ui/skeleton"
import { useToast } from "@/components/ui/use-toast"

export function JobList() {
  const [jobs, setJobs] = useState<Job[] | null>(null)
  const { toast } = useToast()

  const load = () => {
    setJobs(null)
    fetchMyJobs().then(setJobs)
  }

  useEffect(load, [])

  if (!jobs) {
    return <Skeleton className="h-40 w-full" />
  }

  if (jobs.length === 0) {
    return (
      <p className="text-sm text-slate-500">
        No jobs submitted yet
      </p>
    )
  }

  return (
    <Table>
      <TableHeader>
        <TableRow>
          <TableHead>ID</TableHead>
          <TableHead>Status</TableHead>
          <TableHead>Retries</TableHead>
          <TableHead></TableHead>
        </TableRow>
      </TableHeader>

      <TableBody>
        {jobs.map((job) => (
          <TableRow key={job.id}>
            <TableCell>{job.id}</TableCell>
            <TableCell>
              <Badge
                variant={
                  job.status === "COMPLETED"
                    ? "success"
                    : job.status === "FAILED"
                    ? "destructive"
                    : "warning"
                }
              >
                {job.status}
              </Badge>
            </TableCell>
            <TableCell>{job.retry_count}</TableCell>
            <TableCell>
              {job.status === "QUEUED" && (
                <Button
                  size="sm"
                  variant="destructive"
                  onClick={async () => {
                    await cancelJob(job.id)
                    toast({ title: "Job cancelled" })
                    load()
                  }}
                >
                  Cancel
                </Button>
              )}
            </TableCell>
          </TableRow>
        ))}
      </TableBody>
    </Table>
  )
}
```

✔ Cancel only if `QUEUED`
✔ Backend still enforces rule (SRS-safe)
