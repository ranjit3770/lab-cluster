## 📁 `src/services/admin.service.ts`

```ts
import { api } from "./api"

/* ---------- NODES ---------- */
export interface NodeInfo {
  id: number
  node_uid: string
  health_status: string
  total_cpu: number
  free_cpu: number
  total_memory_mb: number
  free_memory_mb: number
}

export async function fetchNodes(): Promise<NodeInfo[]> {
  const { data } = await api.get("/admin/nodes")
  return data
}

/* ---------- USERS ---------- */
export interface CreateUserPayload {
  username: string
  assignment_mode: "GLOBAL" | "NODE"
  max_cpu?: number
  max_memory_mb?: number
  node_quotas?: {
    node_id: number
    cpu: number
    memory_mb: number
  }[]
}

export async function createUser(payload: CreateUserPayload) {
  await api.post("/admin/users", payload)
}

/* ---------- JOBS ---------- */
export interface AdminJob {
  id: number
  user_id: number
  status: string
  retry_count: number
  failure_type?: string
}

export async function fetchAllJobs(
  page: number,
  size: number
): Promise<{ items: AdminJob[]; total: number }> {
  const { data } = await api.get("/admin/jobs", {
    params: { page, size },
  })
  return data
}
```

# 2️⃣ GLOBAL ERROR BOUNDARY (MUST-HAVE)

This prevents **white screens** in production.

---

## 📁 `src/app/ErrorBoundary.tsx`

```tsx
import { Component, ReactNode } from "react"
import { Button } from "@/components/ui/button"

interface Props {
  children: ReactNode
}

interface State {
  hasError: boolean
}

export class ErrorBoundary extends Component<Props, State> {
  state: State = { hasError: false }

  static getDerivedStateFromError() {
    return { hasError: true }
  }

  componentDidCatch(error: Error, info: any) {
    console.error("UI Error:", error, info)
  }

  render() {
    if (this.state.hasError) {
      return (
        <div className="min-h-screen flex flex-col items-center justify-center gap-4">
          <h1 className="text-xl font-semibold">
            Something went wrong
          </h1>
          <p className="text-sm text-slate-500">
            Please refresh or contact admin.
          </p>
          <Button onClick={() => window.location.reload()}>
            Reload
          </Button>
        </div>
      )
    }

    return this.props.children
  }
}
```

## 📁 `src/components/common/Pagination.tsx`

```tsx
import { Button } from "@/components/ui/button"

export function Pagination({
  page,
  total,
  size,
  onChange,
}: {
  page: number
  total: number
  size: number
  onChange: (p: number) => void
}) {
  const pages = Math.ceil(total / size)

  return (
    <div className="flex gap-2 justify-end mt-4">
      <Button
        size="sm"
        disabled={page === 1}
        onClick={() => onChange(page - 1)}
      >
        Prev
      </Button>

      <span className="text-sm px-2 py-1">
        Page {page} of {pages}
      </span>

      <Button
        size="sm"
        disabled={page === pages}
        onClick={() => onChange(page + 1)}
      >
        Next
      </Button>
    </div>
  )
}
```

## 🔄 Admin Jobs Page with Pagination

### `JobsPage.tsx` (updated)

```tsx
import { useEffect, useState } from "react"
import { fetchAllJobs, AdminJob } from "@/services/admin.service"
import { Pagination } from "@/components/common/Pagination"
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
} from "@/components/ui/table"
import { Badge } from "@/components/ui/badge"

export default function JobsPage() {
  const [jobs, setJobs] = useState<AdminJob[]>([])
  const [page, setPage] = useState(1)
  const [total, setTotal] = useState(0)

  const size = 10

  useEffect(() => {
    fetchAllJobs(page, size).then((res) => {
      setJobs(res.items)
      setTotal(res.total)
    })
  }, [page])

  return (
    <>
      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>ID</TableHead>
            <TableHead>User</TableHead>
            <TableHead>Status</TableHead>
            <TableHead>Retries</TableHead>
          </TableRow>
        </TableHeader>

        <TableBody>
          {jobs.map((j) => (
            <TableRow key={j.id}>
              <TableCell>{j.id}</TableCell>
              <TableCell>{j.user_id}</TableCell>
              <TableCell>
                <Badge>{j.status}</Badge>
              </TableCell>
              <TableCell>{j.retry_count}</TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>

      <Pagination
        page={page}
        total={total}
        size={size}
        onChange={setPage}
      />
    </>
  )
}
```
