# 1️⃣ DARK MODE — Toggle + Theme (shadcn way)

shadcn already supports dark mode via **CSS variables + `class="dark"`**.

---

## 1.1 Install theme helper (recommended)

```bash
npm install next-themes
```

> Works perfectly even outside Next.js.

---

## 1.2 Theme Provider

### `src/app/ThemeProvider.tsx`

```tsx
import { ThemeProvider as NextThemeProvider } from "next-themes"
import { ReactNode } from "react"

export function ThemeProvider({ children }: { children: ReactNode }) {
  return (
    <NextThemeProvider
      attribute="class"
      defaultTheme="system"
      enableSystem
    >
      {children}
    </NextThemeProvider>
  )
}
```

---

## 1.3 Wrap App Once

### `main.tsx`

```tsx
import { ThemeProvider } from "@/app/ThemeProvider"

root.render(
  <ThemeProvider>
    <ErrorBoundary>
      <AppRouter />
      <Toaster />
    </ErrorBoundary>
  </ThemeProvider>
)
```

---

## 1.4 Dark Mode Toggle (Topbar)

### `src/components/common/ThemeToggle.tsx`

```tsx
import { Moon, Sun } from "lucide-react"
import { useTheme } from "next-themes"
import { Button } from "@/components/ui/button"

export function ThemeToggle() {
  const { theme, setTheme } = useTheme()

  return (
    <Button
      variant="ghost"
      size="icon"
      onClick={() =>
        setTheme(theme === "dark" ? "light" : "dark")
      }
    >
      {theme === "dark" ? (
        <Sun className="h-4 w-4" />
      ) : (
        <Moon className="h-4 w-4" />
      )}
    </Button>
  )
}
```

Add it to **AdminTopbar** & **UserLayout header**.
