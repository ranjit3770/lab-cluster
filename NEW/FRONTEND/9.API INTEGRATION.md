import { api } from "./api"

export interface JobDTO {
  id: number
  status: string
  retry_count: number
}

export async function submitJob(payload: {
  command: string
  arguments: string[]
}) {
  return api.post("/jobs", payload)
}

export async function getMyJobs(): Promise<JobDTO[]> {
  const { data } = await api.get("/jobs")
  return data
}

export async function cancelJob(id: number) {
  return api.delete(`/jobs/${id}`)
}
```

import { z } from "zod"

export const jobSchema = z.object({
  command: z.string().min(3, "Command required"),
  arguments: z.string().optional(),
})

export type JobForm = z.infer<typeof jobSchema>
```

---

### `JobSubmitCard.tsx` (Final, API-wired)

```tsx
import { useForm } from "react-hook-form"
import { zodResolver } from "@hookform/resolvers/zod"
import { jobSchema, JobForm } from "@/schemas/job.schema"
import { submitJob } from "@/services/jobs.service"
import { useToast } from "@/components/ui/use-toast"

export function JobSubmitCard({ refresh }: { refresh: () => void }) {
  const { toast } = useToast()

  const form = useForm<JobForm>({
    resolver: zodResolver(jobSchema),
    defaultValues: { command: "", arguments: "" },
  })

  const onSubmit = async (data: JobForm) => {
    await submitJob({
      command: data.command,
      arguments: data.arguments?.split(" ") ?? [],
    })
    toast({ title: "Job submitted" })
    form.reset()
    refresh()
  }

  return (
    <form onSubmit={form.handleSubmit(onSubmit)} className="space-y-4">
      <input {...form.register("command")} placeholder="python train.py" />
      <input {...form.register("arguments")} placeholder="--epochs 10" />
      <button type="submit">Submit</button>
    </form>
  )
}
```

✔ Validated
✔ API-safe
✔ No infra exposure

## 5️⃣ USER — Job List (API-wired)

```tsx
import { useEffect, useState } from "react"
import { getMyJobs, cancelJob } from "@/services/jobs.service"

export function JobList() {
  const [jobs, setJobs] = useState<any[]>([])

  const load = async () => {
    setJobs(await getMyJobs())
  }

  useEffect(() => {
    load()
  }, [])

  return (
    <>
      {jobs.map((job) => (
        <div key={job.id}>
          {job.status}
          {job.status === "QUEUED" && (
            <button onClick={() => cancelJob(job.id).then(load)}>
              Cancel
            </button>
          )}
        </div>
      ))}
    </>
  )
}
```

## 6️⃣ Zod Schema — Admin User Create

```ts
// src/schemas/user.schema.ts
import { z } from "zod"

export const userSchema = z.discriminatedUnion("assignment_mode", [
  z.object({
    username: z.string().min(3),
    assignment_mode: z.literal("GLOBAL"),
    max_cpu: z.number().min(1),
    max_memory_mb: z.number().min(512),
  }),
  z.object({
    username: z.string().min(3),
    assignment_mode: z.literal("NODE"),
    node_quotas: z.array(
      z.object({
        node_id: z.number(),
        cpu: z.number().min(1),
        memory_mb: z.number().min(512),
      })
    ),
  }),
])
```
