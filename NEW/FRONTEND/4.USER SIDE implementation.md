# 📝 3️⃣ Job Submit Card (User Entry Point)

### `src/layouts/user/JobSubmitCard.tsx`

```tsx
import { useState } from "react"
import { Card, CardContent, CardHeader, CardTitle } from "@/components/ui/card"
import { Button } from "@/components/ui/button"
import { Input } from "@/components/ui/input"
import { Textarea } from "@/components/ui/textarea"

export function JobSubmitCard() {
  const [command, setCommand] = useState("")
  const [args, setArgs] = useState("")

  const submitJob = () => {
    // POST /jobs
    console.log({
      command,
      arguments: args.split(" ").filter(Boolean),
    })
  }

  return (
    <Card>
      <CardHeader>
        <CardTitle>Submit Job</CardTitle>
      </CardHeader>

      <CardContent className="space-y-3">
        <Input
          placeholder="Command (e.g. python train.py)"
          value={command}
          onChange={(e) => setCommand(e.target.value)}
        />

        <Textarea
          placeholder="Arguments (optional)"
          value={args}
          onChange={(e) => setArgs(e.target.value)}
        />

        <Button onClick={submitJob} className="w-full">
          Submit
        </Button>
      </CardContent>
    </Card>
  )
}
```

🚫 No CPU
🚫 No memory
🚫 No node selection
✔ FR-6 enforced

# 📋 4️⃣ Job List & Tracking

### `src/layouts/user/JobList.tsx`

```tsx
import {
  Table, TableBody, TableCell, TableHead, TableHeader, TableRow,
} from "@/components/ui/table"
import { Badge } from "@/components/ui/badge"

const jobs = [
  {
    id: 201,
    status: "RUNNING",
    retries: 0,
  },
  {
    id: 202,
    status: "RETRY_WAITING",
    retries: 1,
  },
]

export function JobList() {
  return (
    <div className="mt-6">
      <h2 className="text-lg font-semibold mb-2">
        My Jobs
      </h2>

      <Table>
        <TableHeader>
          <TableRow>
            <TableHead>ID</TableHead>
            <TableHead>Status</TableHead>
            <TableHead>Retries</TableHead>
          </TableRow>
        </TableHeader>

        <TableBody>
          {jobs.map((job) => (
            <TableRow key={job.id}>
              <TableCell>{job.id}</TableCell>
              <TableCell>
                <Badge
                  variant={
                    job.status === "COMPLETED"
                      ? "success"
                      : job.status === "FAILED"
                      ? "destructive"
                      : "warning"
                  }
                >
                  {job.status}
                </Badge>
              </TableCell>
              <TableCell>{job.retries}</TableCell>
            </TableRow>
          ))}
        </TableBody>
      </Table>
    </div>
  )
}
```

✔ Job tracking
✔ Retry visibility (count only)
✔ No failure internals leaked
