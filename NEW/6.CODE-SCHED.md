# 🧠 Scheduler Overview (What This Code Guarantees)

* **FIFO per platform**, not per user
* **Hard rule**: max **1 RUNNING job per user**
* **GLOBAL** → any healthy node with capacity
* **NODE** → only assigned healthy nodes with free quota
* **Infra failures** go to `RETRY_WAITING`
* **No blind retries**
* **Audited decisions**

# 📁 `app/scheduler/validator.py`

Validates whether a job *can* run **now**.

```python
from sqlalchemy.orm import Session
from app.models import Job, User, Node, UserNodeQuota
from app.models.enums import JobStatus, NodeHealth, AssignmentMode


class SchedulerValidator:

    @staticmethod
    def user_has_running_job(db: Session, user_id: int) -> bool:
        return db.query(Job).filter(
            Job.user_id == user_id,
            Job.status == JobStatus.RUNNING
        ).first() is not None

    @staticmethod
    def user_is_revoked(user: User) -> bool:
        return user.resource_state.name == "IDLE_REVOKED"

    @staticmethod
    def healthy_nodes(db: Session):
        return db.query(Node).filter(
            Node.health_status == NodeHealth.UP
        ).all()

    @staticmethod
    def eligible_nodes_for_user(db: Session, user: User):
        if user.assignment_mode == AssignmentMode.GLOBAL:
            return SchedulerValidator.healthy_nodes(db)

        # NODE-wise
        return (
            db.query(Node)
            .join(UserNodeQuota, UserNodeQuota.node_id == Node.id)
            .filter(
                UserNodeQuota.user_id == user.id,
                Node.health_status == NodeHealth.UP,
            )
            .all()
        )
```

# 📁 `app/scheduler/placement.py`

Decides **WHERE** the job should run.

```python
from sqlalchemy.orm import Session
from app.models import Node, User, UserNodeQuota
from app.models.enums import AssignmentMode


class PlacementEngine:

    @staticmethod
    def select_node(db: Session, user: User):
        if user.assignment_mode == AssignmentMode.GLOBAL:
            return PlacementEngine._select_global(db, user)

        return PlacementEngine._select_node_bound(db, user)

    @staticmethod
    def _select_global(db: Session, user: User):
        return (
            db.query(Node)
            .filter(
                Node.health_status == "UP",
                Node.free_cpu >= user.max_cpu,
                Node.free_memory_mb >= user.max_memory_mb
            )
            .order_by(Node.free_cpu.desc())
            .first()
        )

    @staticmethod
    def _select_node_bound(db: Session, user: User):
        quotas = (
            db.query(UserNodeQuota)
            .filter(UserNodeQuota.user_id == user.id)
            .all()
        )

        for quota in quotas:
            node = db.query(Node).filter(Node.id == quota.node_id).first()
            if (
                node.health_status == "UP"
                and node.free_cpu >= quota.assigned_cpu
                and node.free_memory_mb >= quota.assigned_memory_mb
            ):
                return node

        return None
```

# 📁 `app/scheduler/queue.py`

FIFO queue operations.

```python
from sqlalchemy.orm import Session
from app.models import SchedulerQueue, Job
from app.models.enums import JobStatus


class SchedulerQueueService:

    @staticmethod
    def enqueue(db: Session, job: Job):
        db.add(SchedulerQueue(job_id=job.id, user_id=job.user_id))
        job.status = JobStatus.QUEUED
        db.commit()

    @staticmethod
    def next_job(db: Session):
        entry = (
            db.query(SchedulerQueue)
            .order_by(SchedulerQueue.enqueued_at.asc())
            .first()
        )
        if not entry:
            return None

        return db.query(Job).filter(Job.id == entry.job_id).first()

    @staticmethod
    def dequeue(db: Session, job_id: int):
        db.query(SchedulerQueue).filter(
            SchedulerQueue.job_id == job_id
        ).delete()
```

# 📁 `app/scheduler/retry_policy.py`

Retry decision logic.

```python
from app.models import Job, User
from app.models.enums import FailureType, JobStatus


class RetryPolicy:

    @staticmethod
    def can_retry(job: Job, user: User) -> bool:
        if job.failure_type != FailureType.INFRA:
            return False

        return job.retry_count < user.retry_limit

    @staticmethod
    def mark_retry(job: Job):
        job.retry_count += 1
        job.status = JobStatus.RETRY_WAITING
```

# 📁 `app/scheduler/scheduler.py`

🔥 **Main Scheduler Brain**

```python
from sqlalchemy.orm import Session
from app.models import Job, User
from app.models.enums import JobStatus
from app.scheduler.queue import SchedulerQueueService
from app.scheduler.validator import SchedulerValidator
from app.scheduler.placement import PlacementEngine
from app.services.audit_service import audit_event
from app.ray_executor.runtime import submit_to_ray


class Scheduler:

    @staticmethod
    def dispatch_next(db: Session):
        job = SchedulerQueueService.next_job(db)
        if not job:
            return None

        user = db.query(User).filter(User.id == job.user_id).first()

        # Hard blocks
        if SchedulerValidator.user_is_revoked(user):
            job.status = JobStatus.FAILED
            audit_event("SYSTEM", "JOB_REJECTED", "JOB", job.id,
                        reason="User revoked")
            db.commit()
            return None

        if SchedulerValidator.user_has_running_job(db, user.id):
            return None  # wait FIFO

        # Placement
        node = PlacementEngine.select_node(db, user)
        if not node:
            return None  # resources unavailable

        # Reserve resources
        job.status = JobStatus.RUNNING
        job.assigned_node_id = node.id

        node.free_cpu -= user.max_cpu
        node.free_memory_mb -= user.max_memory_mb

        SchedulerQueueService.dequeue(db, job.id)
        db.commit()

        # Submit to Ray (async)
        submit_to_ray(job, user, node)

        audit_event(
            "SYSTEM",
            "JOB_DISPATCHED",
            "JOB",
            job.id,
            new_state={"node_id": node.id}
        )

        return job
```

# 🔁 How This Works at Runtime

```
Job submitted
   ↓
Enqueued FIFO
   ↓
Scheduler dispatch tick
   ↓
Validate user + quota
   ↓
Select node (GLOBAL | NODE)
   ↓
Reserve capacity
   ↓
Submit to Ray
```

Failures:

* Node DOWN → job INFRA_FAILED → retry engine
* Retry exhausted → FAILED
