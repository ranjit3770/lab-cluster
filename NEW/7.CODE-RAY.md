# 📁 `app/ray_executor/tasks.py`

**Actual Ray job execution unit**
(Users never see or control this.)

```python
import subprocess
import traceback

def _run_command(command: str, args: list[str]):
    try:
        result = subprocess.run(
            [command] + args,
            capture_output=True,
            text=True,
            check=True
        )
        return {
            "status": "SUCCESS",
            "stdout": result.stdout,
            "stderr": result.stderr
        }
    except subprocess.CalledProcessError as e:
        return {
            "status": "USER_ERROR",
            "stdout": e.stdout,
            "stderr": e.stderr
        }
    except Exception as e:
        return {
            "status": "INFRA_ERROR",
            "error": str(e),
            "traceback": traceback.format_exc()
        }
```

# 📁 `app/ray_executor/node_binding.py`

Creates **node-bound Ray tasks** (NODE mode enforcement).

```python
import ray

def create_ray_task(cpu: int, memory_mb: int, node_uid: str | None):
    resources = {}
    if node_uid:
        resources[f"node:{node_uid}"] = 1

    @ray.remote(
        num_cpus=cpu,
        memory=memory_mb * 1024 * 1024,
        resources=resources
    )
    def task(command: str, args: list[str]):
        from app.ray_executor.tasks import _run_command
        return _run_command(command, args)

    return task
```

# 📁 `app/ray_executor/runtime.py`

🔥 **Executor brain** – called by scheduler.

```python
import ray
from sqlalchemy.orm import Session
from app.db.session import SessionLocal
from app.models import Job, Node, User
from app.models.enums import JobStatus, FailureType
from app.ray_executor.node_binding import create_ray_task
from app.services.audit_service import audit_event
from app.scheduler.retry_policy import RetryPolicy


def submit_to_ray(job: Job, user: User, node: Node):
    db: Session = SessionLocal()

    try:
        # Create Ray task (NODE or GLOBAL)
        ray_task = create_ray_task(
            cpu=user.max_cpu,
            memory_mb=user.max_memory_mb,
            node_uid=node.node_uid if user.assignment_mode.name == "NODE" else None
        )

        ref = ray_task.remote(job.command, job.arguments or [])
        job.ray_task_id = ref.hex()

        db.commit()

        # Async result watcher
        ray.get_runtime_context().worker.core_worker.add_done_callback(
            ref,
            lambda r: _on_ray_complete(r, job.id)
        )

    except Exception as e:
        _handle_infra_failure(job.id, str(e))

    finally:
        db.close()
```

# 📁 `app/ray_executor/result_handler.py`

Handles **Ray completion callbacks**.

```python
from sqlalchemy.orm import Session
from app.db.session import SessionLocal
from app.models import Job, Node, User
from app.models.enums import JobStatus, FailureType
from app.services.audit_service import audit_event
from app.scheduler.retry_policy import RetryPolicy


def _on_ray_complete(ray_result, job_id: int):
    db: Session = SessionLocal()

    job = db.query(Job).filter(Job.id == job_id).first()
    user = db.query(User).filter(User.id == job.user_id).first()
    node = db.query(Node).filter(Node.id == job.assigned_node_id).first()

    # Release node resources
    node.free_cpu += user.max_cpu
    node.free_memory_mb += user.max_memory_mb

    if ray_result["status"] == "SUCCESS":
        job.status = JobStatus.COMPLETED

        audit_event(
            "SYSTEM",
            "JOB_COMPLETED",
            "JOB",
            job.id
        )

    elif ray_result["status"] == "USER_ERROR":
        job.status = JobStatus.FAILED
        job.failure_type = FailureType.USER

        audit_event(
            "SYSTEM",
            "JOB_FAILED_USER",
            "JOB",
            job.id,
            reason=ray_result.get("stderr")
        )

    else:
        # INFRA failure
        job.failure_type = FailureType.INFRA

        if RetryPolicy.can_retry(job, user):
            RetryPolicy.mark_retry(job)

            audit_event(
                "SYSTEM",
                "JOB_RETRY_SCHEDULED",
                "JOB",
                job.id,
                reason="Infra failure"
            )
        else:
            job.status = JobStatus.FAILED

            audit_event(
                "SYSTEM",
                "JOB_FAILED_INFRA",
                "JOB",
                job.id,
                reason="Retry limit exhausted"
            )

    db.commit()
    db.close()
```

# 📁 `app/workers/retry_worker.py`

Periodic **retry engine** (cron / APScheduler).

```python
from sqlalchemy.orm import Session
from app.db.session import SessionLocal
from app.models import Job
from app.models.enums import JobStatus
from app.scheduler.scheduler import Scheduler


def retry_worker():
    db: Session = SessionLocal()

    jobs = db.query(Job).filter(
        Job.status == JobStatus.RETRY_WAITING
    ).all()

    for job in jobs:
        # Re-enqueue safely
        Scheduler.enqueue_for_retry(db, job)

    db.commit()
    db.close()
```

## 🔁 Failure Classification (STRICT)

| Condition                     | Classification |
| ----------------------------- | -------------- |
| Ray worker lost               | INFRA          |
| Node DOWN                     | INFRA          |
| Python exception in user code | USER           |
| Command exit ≠ 0              | USER           |

✔ **Only INFRA retried**
✔ Retry count enforced
✔ Node health rechecked on retry
