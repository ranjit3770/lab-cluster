# 1️⃣ Worker Node Heartbeat Agent

📁 **`scripts/node_heartbeat_agent.py`**

> Runs on **each Ray worker node** (VM / physical machine)

```python
import time
import socket
import psutil
import requests
from datetime import datetime

BACKEND_URL = "http://BACKEND_IP:PORT/internal/nodes/heartbeat"
HEARTBEAT_INTERVAL = 15  # seconds
NODE_UID = socket.gethostname()  # stable identifier


def collect_metrics():
    return {
        "node_uid": NODE_UID,
        "cpu_free": int(psutil.cpu_count() * (1 - psutil.cpu_percent() / 100)),
        "memory_free_mb": int(psutil.virtual_memory().available / (1024 * 1024)),
        "timestamp": datetime.utcnow().isoformat()
    }


def send_heartbeat():
    payload = collect_metrics()
    try:
        r = requests.post(BACKEND_URL, json=payload, timeout=5)
        r.raise_for_status()
    except Exception as e:
        print(f"[HEARTBEAT] Failed: {e}")


if __name__ == "__main__":
    while True:
        send_heartbeat()
        time.sleep(HEARTBEAT_INTERVAL)
```

# 2️⃣ FastAPI Heartbeat API

📁 **`app/api/internal/heartbeat.py`**

```python
from fastapi import APIRouter, HTTPException
from sqlalchemy.orm import Session
from datetime import datetime

from app.db.session import SessionLocal
from app.models import Node
from app.models.enums import NodeHealth
from app.services.audit_service import audit_event

router = APIRouter()


@router.post("/nodes/heartbeat")
def node_heartbeat(payload: dict):
    db: Session = SessionLocal()

    node = (
        db.query(Node)
        .filter(Node.node_uid == payload["node_uid"])
        .first()
    )

    if not node:
        db.close()
        raise HTTPException(status_code=404, detail="Node not registered")

    # Update live metrics
    node.free_cpu = payload["cpu_free"]
    node.free_memory_mb = payload["memory_free_mb"]
    node.last_heartbeat_at = datetime.utcnow()

    # Health recovery
    if node.health_status in [NodeHealth.DOWN, NodeHealth.DEGRADED]:
        node.health_status = NodeHealth.UP
        audit_event(
            actor_type="SYSTEM",
            action_type="NODE_RECOVERED",
            target_type="NODE",
            target_id=node.id
        )

    db.commit()
    db.close()
    return {"status": "OK"}
```

# 3️⃣ Node Health State Machine

📁 **`app/workers/heartbeat_watcher.py`**

> Runs every **N seconds** (cron / APScheduler)

```python
from datetime import datetime, timedelta
from sqlalchemy.orm import Session

from app.db.session import SessionLocal
from app.models import Node, Job
from app.models.enums import NodeHealth, JobStatus, FailureType
from app.services.audit_service import audit_event


HEARTBEAT_TIMEOUT = 30  # seconds
DEGRADED_TIMEOUT = 60  # seconds


def heartbeat_watcher():
    db: Session = SessionLocal()
    now = datetime.utcnow()

    nodes = db.query(Node).all()

    for node in nodes:
        delta = (now - node.last_heartbeat_at).total_seconds()

        if delta > DEGRADED_TIMEOUT and node.health_status != NodeHealth.DOWN:
            node.health_status = NodeHealth.DOWN
            audit_event("SYSTEM", "NODE_DOWN", "NODE", node.id)

            # Fail running jobs on this node
            jobs = db.query(Job).filter(
                Job.assigned_node_id == node.id,
                Job.status == JobStatus.RUNNING
            ).all()

            for job in jobs:
                job.status = JobStatus.RETRY_WAITING
                job.failure_type = FailureType.INFRA

                audit_event(
                    "SYSTEM",
                    "JOB_INFRA_FAILED",
                    "JOB",
                    job.id,
                    reason="Node heartbeat lost"
                )

        elif delta > HEARTBEAT_TIMEOUT:
            if node.health_status != NodeHealth.DEGRADED:
                node.health_status = NodeHealth.DEGRADED
                audit_event("SYSTEM", "NODE_DEGRADED", "NODE", node.id)

    db.commit()
    db.close()
```

# 4️⃣ Scheduler Integration (Automatic)

### What Happens Automatically

| Event            | Effect                           |
| ---------------- | -------------------------------- |
| Node → DEGRADED  | No new jobs scheduled            |
| Node → DOWN      | Running jobs marked INFRA_FAILED |
| Node → UP        | Scheduler resumes placement      |
| Job INFRA_FAILED | Retry engine evaluates           |

**No manual intervention required.**
