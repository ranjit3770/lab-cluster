1. ✅ **Super Admin user**

* normal users
* jobs
* audit logs

## 1️⃣ Seed Super Admin

📁 `scripts/seed_admin.py`

```python
from sqlalchemy.orm import Session
from passlib.context import CryptContext

from app.db.session import SessionLocal
from app.models.user import User
from app.models.enums import UserRole, AssignmentMode, ResourceState

pwd_context = CryptContext(schemes=["bcrypt"], deprecated="auto")


def seed_admin():
    db: Session = SessionLocal()

    admin_username = "admin"
    admin_password = "ChangeMeImmediately!"  # override via env in prod

    exists = db.query(User).filter(User.username == admin_username).first()
    if exists:
        print("ℹ️ Super admin already exists")
        db.close()
        return

    admin = User(
        username=admin_username,
        password_hash=pwd_context.hash(admin_password),
        role=UserRole.SUPER_ADMIN,
        assignment_mode=AssignmentMode.GLOBAL,
        max_cpu=0,
        max_memory_mb=0,
        retry_limit=0,
        resource_state=ResourceState.HEALTHY,
    )

    db.add(admin)
    db.commit()
    db.close()

    print("✅ Super admin created")
```

🔐 **Production tip**

* Load password from environment variable
* Force password change on first login

## 2️⃣ Seed Nodes

📁 `scripts/seed_nodes.py`

```python
from sqlalchemy.orm import Session
from app.db.session import SessionLocal
from app.models.node import Node
from app.models.enums import NodeHealth
from datetime import datetime


# Example cluster definition
NODES = [
    {
        "node_uid": "node-1",
        "total_cpu": 8,
        "total_memory_mb": 32768,
    },
    {
        "node_uid": "node-2",
        "total_cpu": 16,
        "total_memory_mb": 65536,
    },
]


def seed_nodes():
    db: Session = SessionLocal()

    for node_def in NODES:
        node = (
            db.query(Node)
            .filter(Node.node_uid == node_def["node_uid"])
            .first()
        )

        if node:
            print(f"ℹ️ Node {node.node_uid} already exists")
            continue

        node = Node(
            node_uid=node_def["node_uid"],
            total_cpu=node_def["total_cpu"],
            total_memory_mb=node_def["total_memory_mb"],
            free_cpu=node_def["total_cpu"],
            free_memory_mb=node_def["total_memory_mb"],
            health_status=NodeHealth.UP,
            last_heartbeat_at=datetime.utcnow(),
        )

        db.add(node)
        print(f"✅ Node seeded: {node.node_uid}")

    db.commit()
    db.close()
```

📌 **Important**

* `free_*` initialized = `total_*`
* Heartbeat agent will take over after boot

## 3️⃣ Seed System Settings

📁 `scripts/seed_settings.py`

```python
from sqlalchemy.orm import Session
from app.db.session import SessionLocal
from app.models.system_settings import SystemSetting


DEFAULT_SETTINGS = {
    "IDLE_DAYS": "30",
    "HEARTBEAT_TIMEOUT": "30",          # seconds
    "DEGRADED_TIMEOUT": "60",            # seconds
    "SCHEDULER_TICK_SECONDS": "5",
    "MAX_RETRY_WINDOW_MINUTES": "60",
}


def seed_settings():
    db: Session = SessionLocal()

    for key, value in DEFAULT_SETTINGS.items():
        setting = (
            db.query(SystemSetting)
            .filter(SystemSetting.setting_key == key)
            .first()
        )

        if setting:
            print(f"ℹ️ Setting {key} already exists")
            continue

        db.add(SystemSetting(
            setting_key=key,
            setting_value=value
        ))

        print(f"✅ Setting added: {key}={value}")

    db.commit()
    db.close()
```
