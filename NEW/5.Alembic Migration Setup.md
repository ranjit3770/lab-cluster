# 🧬 Alembic Migration Setup

---

## 1️⃣ Install & Initialize Alembic

```bash
pip install alembic
alembic init alembic
```

Project structure will become:

```text
alembic/
├── versions/
├── env.py
├── script.py.mako
alembic.ini
```

---

## 2️⃣ Configure `alembic.ini`

```ini
sqlalchemy.url = mysql+mysqlconnector://USER:PASSWORD@HOST:PORT/DBNAME
```

👉 **Do not hardcode in prod** (we’ll override in `env.py`).

## 3️⃣ Update `alembic/env.py` (IMPORTANT)

```python
from logging.config import fileConfig
from sqlalchemy import engine_from_config, pool
from alembic import context

from app.db.session import Base
from app.core.config import settings

# Import ALL models so Alembic sees them
from app.models.user import User
from app.models.node import Node
from app.models.user_node_quota import UserNodeQuota
from app.models.job import Job
from app.models.resource_request import ResourceRequest
from app.models.audit_log import AuditLog
from app.models.scheduler_queue import SchedulerQueue
from app.models.system_settings import SystemSetting

config = context.config

config.set_main_option(
    "sqlalchemy.url",
    f"mysql+mysqlconnector://{settings.DB_USER}:"
    f"{settings.DB_PASSWORD}@{settings.DB_HOST}:"
    f"{settings.DB_PORT}/{settings.DB_NAME}"
)

if config.config_file_name is not None:
    fileConfig(config.config_file_name)

target_metadata = Base.metadata


def run_migrations_offline():
    context.configure(
        url=config.get_main_option("sqlalchemy.url"),
        target_metadata=target_metadata,
        literal_binds=True,
        dialect_opts={"paramstyle": "named"},
    )
    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online():
    connectable = engine_from_config(
        config.get_section(config.config_ini_section),
        prefix="sqlalchemy.",
        poolclass=pool.NullPool,
    )

    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=target_metadata,
            compare_type=True
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()
```
